```python
# Tabla gap POST-migraciones — misma logica que PRE
filas_gap_post = []
gaps_residuales = []

for mapping in MAPPINGS:
    if "OUT-OF-SCOPE" in mapping.notas:
        continue
    
    xlsx_candidates = list((Path("../data/proyectos_individuales")).glob(f"*{mapping.expediente}*.xlsx"))
    if not xlsx_candidates:
        continue
    
    bc3 = cargar_bc3(xlsx_candidates[0])
    pem_bc3_total = sum(agrupar_por_capitulo(bc3).values())
    
    try:
        params_dict = derivar_parametros(bc3, mapping, precios_base_post)
        parametros = ParametrosProyecto(**params_dict)
        resultado = calcular_presupuesto(parametros, precios_base_post)
        pem_licitaia = resultado["pem"]
        gap_eur, gap_pct = comparar_gap(pem_bc3_total, pem_licitaia)
        
        es_parcial = "parcial" in mapping.notas.lower()
        dentro_tolerancia = abs(gap_pct) <= TOLERANCIA_GLOBAL
        
        tipo_gap_final = "ACEPTABLE" if dentro_tolerancia else (
            "formula gap" if abs(gap_pct) <= 0.35 else "scope gap"
        )
        
        if not dentro_tolerancia and not es_parcial:
            gaps_residuales.append({
                "expediente": mapping.expediente,
                "gap_pct": gap_pct,
                "tipo": tipo_gap_final,
                "causa": "Ver analisis en celda siguiente",
            })
        
        filas_gap_post.append({
            "expediente": mapping.expediente,
            "nombre": mapping.nombre_obra,
            "pem_bc3": pem_bc3_total,
            "pem_licitaia": pem_licitaia,
            "gap_eur": gap_eur,
            "gap_pct": gap_pct,
            "tipo_gap": tipo_gap_final,
            "parcial": es_parcial,
        })
    
    except ValueError as e:
        gaps_residuales.append({
            "expediente": mapping.expediente,
            "gap_pct": None,
            "tipo": "catalog gap residual",
            "causa": str(e)[:120],
        })
        print(f"[CATALOG GAP RESIDUAL] {mapping.expediente}: {e}")

# Imprimir tabla POST
print(f"\n{'='*90}")
print(f"TABLA GAP POST-MIGRACIONES")
print(f"{'='*90}")
print(f"{'Expediente':<15} {'Nombre':<20} {'PEM BC3':>12} {'PEM LicitaIA':>14} {'gap %':>8} {'Tipo'}")
print("-" * 90)
for f in filas_gap_post:
    parcial_flag = "(parcial)" if f["parcial"] else ""
    print(f"{f['expediente']:<15} {f['nombre']:<20} {f['pem_bc3']:>12,.2f} "
          f"{f['pem_licitaia']:>14,.2f} {f['gap_pct']*100:>7.1f}% "
          f"{f['tipo_gap']} {parcial_flag}")
```
