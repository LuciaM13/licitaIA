La funcion recibe `bc3: list[PartidaBC3]`, `mapping: MappingObra`, y `precios_base: dict`

Devuelve un `dict` con todos los campos de `ParametrosProyecto` para que el notebook

**Estructura de la implementacion:**

```python
def derivar_parametros(
    bc3: list[PartidaBC3],
    mapping: MappingObra,
    precios_base: dict,
) -> dict:
    """Deriva los parametros de ParametrosProyecto desde el BC3 y el mapping curado.
    
    Devuelve un dict con todos los campos necesarios.
    Ningún valor se inventa: si el mapping no tiene un selector para un campo,
    se usa el default seguro (0.0 o el item placeholder del catalogo).
    """
    
    def _resolver(campo) -> float:
        """Resuelve PartidaSelector -> float, o devuelve float directamente."""
        if campo is None:
            return 0.0
        if isinstance(campo, PartidaSelector):
            return resolver_selector(bc3, campo)
        return float(campo)
    
    def _buscar_item(catalogo: list[dict], label: str) -> dict | None:
        """Busca un item por label exacto en una lista de items del catalogo."""
        return next((i for i in catalogo if i["label"] == label), None)
    
    def _primer_item(catalogo: list[dict]) -> dict:
        """Devuelve el primer item del catalogo (para campos con m2=0 / m=0)."""
        if not catalogo:
            raise ValueError(f"Catalogo vacio — no hay items de referencia")
        return catalogo[0]
    
    # --- Tuberias ABA ---
    cat_aba = precios_base.get("catalogo_aba", [])
    cat_san = precios_base.get("catalogo_san", [])
    acerados_aba = precios_base.get("acerados_aba", [])
    acerados_san = precios_base.get("acerados_san", [])
    calzadas = precios_base.get("calzadas", [])
    acometidas_aba = precios_base.get("acometidas_aba", [])
    acometidas_san = precios_base.get("acometidas_san", [])
    
    # Tuberia: buscar el item del catalogo por label segun el expediente.
    # El label viene del MappingObra o se infiere del diametro y material.
    # Para obras in-scope, el mapping debe tener un campo aba_label o san_label.
    # Si no se puede resolver, lanzar ValueError con mensaje claro (catalog gap).
    aba_item = None
    if mapping.es_aba and hasattr(mapping, 'aba_label') and mapping.aba_label:
        aba_item = _buscar_item(cat_aba, mapping.aba_label)
        if aba_item is None:
            raise ValueError(
                f"[CATALOG GAP] Label '{mapping.aba_label}' no encontrado en catalogo ABA "
                f"para {mapping.expediente}. Agregar migracion antes de derivar_parametros."
            )
    
    san_item = None
    if mapping.es_san and hasattr(mapping, 'san_label') and mapping.san_label:
        san_item = _buscar_item(cat_san, mapping.san_label)
        if san_item is None:
            raise ValueError(
                f"[CATALOG GAP] Label '{mapping.san_label}' no encontrado en catalogo SAN "
                f"para {mapping.expediente}."
            )
    
    # Pavimentacion — usar primer item si m2 derivado es 0.0
    pav_aba_acerado_m2 = _resolver(mapping.m2_acerado_aba)
    pav_aba_acerado_item = (
        _buscar_item(acerados_aba, mapping.acerado_aba_label)
        if hasattr(mapping, 'acerado_aba_label') and mapping.acerado_aba_label
        else _primer_item(acerados_aba)
    )
    
    pav_aba_calzada_m2 = _resolver(mapping.m2_calzada_aba)
    pav_aba_calzada_item = (
        _buscar_item(calzadas, mapping.calzada_aba_label)
        if hasattr(mapping, 'calzada_aba_label') and mapping.calzada_aba_label
        else _primer_item(calzadas)
    )
    
    pav_san_calzada_m2 = _resolver(mapping.m2_calzada_san)
    pav_san_calzada_item = (
        _buscar_item(calzadas, mapping.calzada_san_label)
        if hasattr(mapping, 'calzada_san_label') and mapping.calzada_san_label
        else _primer_item(calzadas)
    )
    
    pav_san_acera_m2 = _resolver(mapping.m2_acerado_san)
    pav_san_acera_item = (
        _buscar_item(acerados_san, mapping.acerado_san_label)
        if hasattr(mapping, 'acerado_san_label') and mapping.acerado_san_label
        else _primer_item(acerados_san)
    )
    
    return {
        "aba_item": aba_item,
        "aba_longitud_m": _resolver(mapping.longitud_zanja_aba_m),
        "aba_profundidad_m": mapping.profundidad_aba_m or 0.0,
        "san_item": san_item,
        "san_longitud_m": _resolver(mapping.longitud_zanja_san_m),
        "san_profundidad_m": mapping.profundidad_san_m or 0.0,
        "pav_aba_acerado_m2": pav_aba_acerado_m2,
        "pav_aba_acerado_item": pav_aba_acerado_item,
        "pav_aba_bordillo_m": 0.0,         # no modelado individualmente en BC3 simple
        "pav_aba_bordillo_item": _primer_item(acerados_aba),
        "pav_aba_calzada_m2": pav_aba_calzada_m2,
        "pav_aba_calzada_item": pav_aba_calzada_item,
        "pav_san_calzada_m2": pav_san_calzada_m2,
        "pav_san_calzada_item": pav_san_calzada_item,
        "pav_san_acera_m2": pav_san_acera_m2,
        "pav_san_acera_item": pav_san_acera_item,
        "material_demo_bordillo_aba": getattr(mapping, 'material_demo_bordillo_aba', 'granitico'),
        "material_demo_acerado_aba": getattr(mapping, 'material_demo_acerado_aba', 'losa_hidraulica'),
        "material_demo_calzada_aba": getattr(mapping, 'material_demo_calzada_aba', 'aglomerado'),
        "acometidas_aba_n": int(_resolver(mapping.n_acometidas_aba)),
        "acometidas_san_n": int(_resolver(mapping.n_acometidas_san)),
        "pct_manual": mapping.pct_manual,
        "conduccion_provisional_m": mapping.conduccion_provisional_m,
        "pct_seguridad": mapping.pct_seguridad,
        "pct_gestion": mapping.pct_gestion,
        "pct_servicios_afectados": getattr(mapping, 'pct_servicios_afectados', 0.0),
        "desmontaje_tipo": "fibrocemento" if mapping.tiene_fibrocemento else "normal",
        "pozos_existentes_aba": getattr(mapping, 'pozos_existentes_aba', 'none'),
        "pozos_existentes_san": getattr(mapping, 'pozos_existentes_san', 'none'),
        "imbornales_tipo": getattr(mapping, 'imbornales_tipo', 'none'),
        "imbornales_nuevo_label": getattr(mapping, 'imbornales_nuevo_label', ''),
    }
```

```python
# Gap analysis PRE-migraciones
# Para cada obra in-scope: derivar parametros -> calcular_presupuesto -> tabla gap

TOLERANCIA_GLOBAL = 0.15   # AACE Class 5 +/- 15%
TOLERANCIA_CAPITULO = 0.20  # +/- 20% por capitulo

filas_gap = []
catalog_gaps_identificados = []

for mapping in MAPPINGS:
    if "OUT-OF-SCOPE" in mapping.notas:
        continue
    
    xlsx_candidates = list((Path("../data/proyectos_individuales")).glob(f"*{mapping.expediente}*.xlsx"))
    if not xlsx_candidates:
        print(f"[AVISO] No se encuentra xlsx para {mapping.expediente} — omitiendo")
        continue
    
    bc3 = cargar_bc3(xlsx_candidates[0])
    pem_bc3_total = sum(agrupar_por_capitulo(bc3).values())
    
    try:
        params_dict = derivar_parametros(bc3, mapping, precios_base)
        parametros = ParametrosProyecto(**params_dict)
        resultado = calcular_presupuesto(parametros, precios_base)
        pem_licitaia = resultado["pem"]
        gap_eur, gap_pct = comparar_gap(pem_bc3_total, pem_licitaia)
        
        # Gap global
        tipo_global = "ACEPTABLE" if abs(gap_pct) <= TOLERANCIA_GLOBAL else "EXCEDE"
        filas_gap.append({
            "expediente": mapping.expediente,
            "nombre": mapping.nombre_obra,
            "capitulo": "TOTAL",
            "pem_bc3": pem_bc3_total,
            "pem_licitaia": pem_licitaia,
            "gap_eur": gap_eur,
            "gap_pct": gap_pct,
            "tipo_gap": tipo_global,
            "clasificacion": "in-scope" if "parcial" not in mapping.notas.lower() else "parcial",
        })
        
        # Gap por capitulo de LicitaIA
        for cap_nombre, cap_data in resultado["capitulos"].items():
            sub_licitaia = cap_data["subtotal"]
            # El PEM BC3 por capitulo es una aproximacion — mapear capitulo BC3 a LicitaIA requiere
            # conocimiento de la obra. Para la tabla PRE-migraciones, reportar el subtotal LicitaIA
            # contra el importe BC3 del capitulo equivalente si se puede mapear.
            filas_gap.append({
                "expediente": mapping.expediente,
                "nombre": mapping.nombre_obra,
                "capitulo": cap_nombre,
                "pem_bc3": None,  # mapeo BC3 -> cap LicitaIA en siguiente celda
                "pem_licitaia": sub_licitaia,
                "gap_eur": None,
                "gap_pct": None,
                "tipo_gap": "—",
                "clasificacion": "",
            })
    
    except ValueError as e:
        msg = str(e)
        if "[CATALOG GAP]" in msg or "no se resolvio" in msg:
            catalog_gaps_identificados.append({
                "expediente": mapping.expediente,
                "error": msg,
            })
            print(f"[CATALOG GAP] {mapping.expediente}: {msg}")
        else:
            print(f"[ERROR] {mapping.expediente}: {msg}")
        continue

# Imprimir tabla gap global
print(f"\n{'='*80}")
print(f"TABLA GAP PRE-MIGRACIONES — NIVEL GLOBAL")
print(f"{'='*80}")
print(f"{'Expediente':<15} {'Nombre':<20} {'PEM BC3':>12} {'PEM LicitaIA':>14} {'gap EUR':>10} {'gap %':>8} {'Tipo'}")
print("-" * 85)
for f in filas_gap:
    if f["capitulo"] == "TOTAL":
        print(f"{f['expediente']:<15} {f['nombre']:<20} {f['pem_bc3']:>12,.2f} "
              f"{f['pem_licitaia']:>14,.2f} {f['gap_eur']:>10,.2f} "
              f"{f['gap_pct']*100:>7.1f}% {f['tipo_gap']}")

if catalog_gaps_identificados:
    print(f"\nCATALOG GAPS identificados ({len(catalog_gaps_identificados)}):")
    for cg in catalog_gaps_identificados:
        print(f"  {cg['expediente']}: {cg['error'][:100]}")
```
