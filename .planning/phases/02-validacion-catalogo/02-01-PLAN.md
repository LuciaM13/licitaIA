   ```python
   def agrupar_por_capitulo(partidas: list[PartidaBC3]) -> dict[str, float]:
       """Devuelve {capitulo: importe_total}. Clave es zero-padded ("01", "02")."""
       totales: dict[str, float] = {}
       for p in partidas:
           totales[p.capitulo] = totales.get(p.capitulo, 0.0) + p.importe
       return totales
   ```

6. **Funcion seleccionar_partidas** (usada por derivar_parametros):
   ```python
   def seleccionar_partidas(
       bc3: list[PartidaBC3],
       selector: PartidaSelector,
   ) -> list[PartidaBC3]:
       """Filtra bc3 por capitulo (si se especifica) y por substring de descripcion."""
       resultado = bc3
       if selector.capitulo is not None:
           resultado = [p for p in resultado if p.capitulo == selector.capitulo]
       if selector.descripcion_substr is not None:
           substr = selector.descripcion_substr.lower()
           resultado = [p for p in resultado if substr in p.descripcion.lower()]
       if selector.codigo_exacto is not None:
           resultado = [p for p in resultado if p.codigo == selector.codigo_exacto]
       return resultado
   ```

7. **Funcion resolver_selector** (devuelve float; suma si agregar=True):
   ```python
   def resolver_selector(
       bc3: list[PartidaBC3],
       selector: PartidaSelector,
   ) -> float:
       """Resuelve un PartidaSelector a un valor float (medicion o importe agregado)."""
       partidas = seleccionar_partidas(bc3, selector)
       if not partidas:
           return 0.0
       if selector.agregar:
           return sum(p.medicion for p in partidas)
       return partidas[0].medicion  # primera coincidencia si agregar=False
   ```

9. **Funcion comparar_gap** (comparador de PEM global y por capitulo):
   ```python
   def comparar_gap(
       pem_real: float,
       pem_licitaia: float,
   ) -> tuple[float, float]:
       """Devuelve (gap_eur, gap_pct).
       
       gap_pct = (pem_real - pem_licitaia) / pem_real
       Positivo: LicitaIA subestima. Negativo: LicitaIA sobreestima.
       """
       if pem_real == 0:
           return 0.0, 0.0
       gap_eur = pem_real - pem_licitaia
       gap_pct = gap_eur / pem_real
       return gap_eur, gap_pct
   ```
